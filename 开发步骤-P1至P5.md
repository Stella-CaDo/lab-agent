# 智能实验室预约系统 · P1–P5 详细开发步骤

> 项目：`E:\Development\Vue\lab-agent\frontend`（Vue 3 + Vite 8 + Element Plus + vue-router + pinia + axios + echarts）
> 用法：**一次只做一个阶段**，验收全过再做下一阶段；每阶段结束都要手点一遍并 `git commit`。
> 约定：路径相对 `frontend/`；每个文件都标了**新建**或**替换**。

## 0. 阶段总览

| 阶段 | 内容 | 预计 | 完成后能演示 | 对应作业考点 |
|---|---|---|---|---|
| **P1** | mock 接口 + axios 封装 + pinia store | 1 小时 | 接口返回真实数据 | 网络请求、状态管理 |
| **P2** | 实验室管理（搜索/分页/增删改） | 3 小时 | 一条完整 CRUD 跑通 | 数据绑定、组件、增删改查 |

| **P4** | 仪表盘 4 指标 + 2 图；统计页 4 图 | 3 小时 | 首屏有图有数 | 多页面、图表统计 |
| **P5** | 设备页、用户页、登录页、404、构建 | 2 小时 | 完整交付 | 路由守卫、工程化 |

**每阶段固定流程**

1. 按文档落地文件（新建 / 替换）
2. 改了 `vite.config.js` 必须重启 dev server
3. 对着"验收清单"逐条手点
4. `git add -A` → `git commit -m "..."`
5. 回一句"P? 好了"，再进入下一阶段

```js
// ============================================================
// 自写 Vite mock 中间件（零额外依赖）
// 只在 vite dev 生效（apply: 'serve'），不进入生产构建
// 统一响应：{ code: 0, msg, data }；业务失败 { code: 1, msg }
// ============================================================
import { labs, users, devices, reservations, nextId, offsetDate, TIME_SLOTS } from './db.js'

// 接口层不做鉴权，登录守卫放在前端路由（P5）；改成 true 可开启严格鉴权
const REQUIRE_AUTH = false

/* ---------------- 基础工具 ---------------- */
const ok = (res, data = null, msg = 'ok') => reply(res, 200, { code: 0, msg, data })
const fail = (res, msg = '参数错误', status = 200) => reply(res, status, { code: 1, msg, data: null })

function reply(res, status, payload) {
  res.statusCode = status
  res.setHeader('Content-Type', 'application/json; charset=utf-8')
  res.end(JSON.stringify(payload))
}

/** 读取 JSON 请求体（Connect 中间件里 req 就是 Node 的 IncomingMessage） */
function readJson(req) {
  return new Promise((resolve, reject) => {
    let raw = ''
    req.on('data', (chunk) => (raw += chunk))
    req.on('end', () => {
      if (!raw) return resolve({})
      try {
        resolve(JSON.parse(raw))
      } catch {
        reject(new Error('请求体不是合法 JSON'))
      }
    })
    req.on('error', reject)
  })
}

/** 内存分页 */
function paginate(list, page = 1, size = 10) {
  const p = Math.max(1, Number(page) || 1)
  const s = Math.max(1, Number(size) || 10)
  return { list: list.slice((p - 1) * s, p * s), total: list.length, page: p, size: s }
}

/** 时段重叠判定：新开始 < 老结束 且 老开始 < 新结束 */
const isOverlap = (aStart, aEnd, bStart, bEnd) => aStart < bEnd && bStart < aEnd

/** 给预约补上实验室名 / 用户名，表格直接展示 */
function withNames(r) {
  const lab = labs.find((l) => l.id === r.labId)
  const user = users.find((u) => u.id === r.userId)
  return { ...r, labName: lab ? lab.name : '已删除实验室', userName: user ? user.name : '未知用户' }
}

/* ---------------- 主处理函数 ---------------- */
export async function handleApi(req, res, url) {
  const path = url.pathname.replace(/\/+$/, '')
  const method = req.method.toUpperCase()

  /* ===== 登录（P5 的登录页调用） ===== */
  if (path === '/api/auth/login' && method === 'POST') {
    const body = await readJson(req)
    const user = users.find((u) => u.account === body.account)
    if (!user || user.password !== body.password) return fail(res, '账号或密码错误')
    if (user.status !== 'active') return fail(res, '账号已被禁用')
    const { password, ...safeUser } = user
    return ok(res, { token: 'mock-token-' + user.id, user: safeUser }, '登录成功')
  }

  /* ===== 实验室 CRUD ===== */
  if (path === '/api/labs' && method === 'GET') {
    const keyword = (url.searchParams.get('keyword') || '').trim()
    const building = url.searchParams.get('building') || ''
    const status = url.searchParams.get('status') || ''
    const filtered = labs.filter((l) => {
      const hit = !keyword || l.name.includes(keyword) || l.room.includes(keyword) || l.building.includes(keyword)
      return hit && (!building || l.building === building) && (!status || l.status === status)
    })
    return ok(res, paginate(filtered, url.searchParams.get('page'), url.searchParams.get('size')))
  }

  if (path === '/api/labs' && method === 'POST') {
    const body = await readJson(req)
    const err = validateLab(body)
    if (err) return fail(res, err)
    const lab = { id: nextId(labs), status: 'available', remark: '', ...body }
    labs.push(lab)
    return ok(res, lab, '新增成功')
  }

  let m = /^\/api\/labs\/(\d+)$/.exec(path)
  if (m) {
    const id = Number(m[1])
    const idx = labs.findIndex((l) => l.id === id)
    if (idx === -1) return fail(res, '实验室不存在', 404)
    if (method === 'GET') return ok(res, labs[idx])
    if (method === 'PUT') {
      const body = await readJson(req)
      const err = validateLab(body)
      if (err) return fail(res, err)
      labs[idx] = { ...labs[idx], ...body, id }
      return ok(res, labs[idx], '修改成功')
    }
    if (method === 'DELETE') {
      const used = reservations.some((r) => r.labId === id && r.status === 'pending')
      if (used) return fail(res, '该实验室还有待审核预约，不能删除')
      labs.splice(idx, 1)
      return ok(res, null, '删除成功')
    }
  }


  /* ===== 预约 ===== */
  if (path === '/api/reservations' && method === 'GET') {
    const status = url.searchParams.get('status') || ''
    const date = url.searchParams.get('date') || ''
    const labId = url.searchParams.get('labId') || ''
    const filtered = reservations
      .filter((r) => (!status || r.status === status) && (!date || r.date === date) && (!labId || String(r.labId) === labId))
      .slice()
      .sort((a, b) => (a.date === b.date ? (a.startTime < b.startTime ? 1 : -1) : a.date < b.date ? 1 : -1))
      .map(withNames)
    return ok(res, paginate(filtered, url.searchParams.get('page'), url.searchParams.get('size')))
  }

  if (path === '/api/reservations' && method === 'POST') {
    const body = await readJson(req)
    const lab = labs.find((l) => l.id === Number(body.labId))
    if (!lab) return fail(res, '实验室不存在')
    if (lab.status !== 'available') return fail(res, '该实验室当前不可预约')
    if (!body.date || !body.startTime || !body.endTime) return fail(res, '请选择预约日期和时段')
    if (body.date < offsetDate(0)) return fail(res, '不能预约过去的日期')
    if (body.startTime >= body.endTime) return fail(res, '结束时间必须晚于开始时间')
    if (body.startTime < lab.openTime || body.endTime > lab.closeTime) {
      return fail(res, '预约时段超出实验室开放时间 ' + lab.openTime + '-' + lab.closeTime)
    }
    if (Number(body.attendees) > lab.capacity) return fail(res, '预约人数不能超过实验室容量 ' + lab.capacity)
    // 冲突校验：同实验室 + 同日期 + 时段重叠 + 状态有效
    const conflict = reservations.find(
      (r) =>
        r.labId === lab.id &&
        r.date === body.date &&
        (r.status === 'pending' || r.status === 'approved') &&
        isOverlap(body.startTime, body.endTime, r.startTime, r.endTime)
    )
    if (conflict) return fail(res, '时段冲突：' + conflict.startTime + '-' + conflict.endTime + ' 已被占用')
    const record = {
      id: nextId(reservations),
      labId: lab.id,
      userId: Number(body.userId) || 1,
      date: body.date,
      startTime: body.startTime,
      endTime: body.endTime,
      purpose: body.purpose || '课程实验',
      attendees: Number(body.attendees) || 1,
      status: 'pending',
      createdAt: offsetDate(0) + ' ' + new Date().toTimeString().slice(0, 5)
    }
    reservations.push(record)
    return ok(res, withNames(record), '预约提交成功，等待审核')
  }

  m = /^\/api\/reservations\/(\d+)\/status$/.exec(path)
  if (m && method === 'PUT') {
    const target = reservations.find((r) => r.id === Number(m[1]))
    if (!target) return fail(res, '预约不存在', 404)
    const body = await readJson(req)
    // 状态机：待审核 转 通过/驳回；已通过 转 取消
    const allowMap = { pending: ['approved', 'rejected'], approved: ['cancelled'] }
    if (!(allowMap[target.status] || []).includes(body.status)) return fail(res, '当前状态不允许这样流转')
    target.status = body.status
    return ok(res, withNames(target), '状态已更新')
  }

  m = /^\/api\/reservations\/(\d+)$/.exec(path)
  if (m && method === 'DELETE') {
    const idx = reservations.findIndex((r) => r.id === Number(m[1]))
    if (idx === -1) return fail(res, '预约不存在', 404)
    reservations.splice(idx, 1)
    return ok(res, null, '删除成功')
  }

  /* ===== 设备 ===== */
  if (path === '/api/devices' && method === 'GET') {
    const keyword = (url.searchParams.get('keyword') || '').trim()
    const labId = url.searchParams.get('labId') || ''
    const filtered = devices
      .filter((d) => (!labId || String(d.labId) === labId) && (!keyword || d.name.includes(keyword) || d.model.includes(keyword)))
      .map((d) => ({ ...d, labName: (labs.find((l) => l.id === d.labId) || {}).name || '-' }))
    return ok(res, paginate(filtered, url.searchParams.get('page'), url.searchParams.get('size')))
  }

  /* ===== 用户（只读） ===== */
  if (path === '/api/users' && method === 'GET') {
    const list = users.map((u) => {
      const item = { ...u }
      delete item.password
      return item
    })
    return ok(res, paginate(list, url.searchParams.get('page'), url.searchParams.get('size')))
  }

  /* ===== 统计 ===== */
  if (path === '/api/stats/overview' && method === 'GET') {
    const today = offsetDate(0)
    const usedToday = new Set(
      reservations
        .filter((r) => r.date === today && r.status !== 'rejected' && r.status !== 'cancelled')
        .map((r) => r.labId + '|' + r.startTime)
    ).size
    return ok(res, {
      labCount: labs.length,
      labAvailable: labs.filter((l) => l.status === 'available').length,
      todayReservations: reservations.filter((r) => r.date === today).length,
      pendingCount: reservations.filter((r) => r.status === 'pending').length,
      avgUsage: Math.round((usedToday / (labs.length * TIME_SLOTS.length)) * 1000) / 10
    })
  }

  if (path === '/api/stats/trend' && method === 'GET') {
    const range = Number(url.searchParams.get('range')) || 7
    const days = []
    for (let i = range - 1; i >= 0; i--) {
      const date = offsetDate(-i)
      days.push({ date: date.slice(5), count: reservations.filter((r) => r.date === date).length })
    }
    return ok(res, days)
  }

  if (path === '/api/stats/lab-usage' && method === 'GET') {
    const list = labs
      .map((l) => ({ name: l.name, count: reservations.filter((r) => r.labId === l.id && r.status === 'approved').length }))
      .sort((a, b) => b.count - a.count)
    return ok(res, list)
  }

  if (path === '/api/stats/status-ratio' && method === 'GET') {
    const nameMap = { pending: '待审核', approved: '已通过', rejected: '已驳回', cancelled: '已取消' }
    const list = Object.keys(nameMap).map((k) => ({
      name: nameMap[k],
      value: reservations.filter((r) => r.status === k).length
    }))
    return ok(res, list)
  }

  if (path === '/api/stats/heatmap' && method === 'GET') {
    const data = []
    for (let day = 0; day < 7; day++) {
      for (let slot = 0; slot < TIME_SLOTS.length; slot++) {
        const value = reservations.filter((r) => {
          const d = new Date(r.date + 'T00:00:00')
          return d.getDay() === day && r.startTime === TIME_SLOTS[slot] && r.status !== 'rejected' && r.status !== 'cancelled'
        }).length
        data.push([day, slot, value])
      }
    }
    return ok(res, { days: ['周日', '周一', '周二', '周三', '周四', '周五', '周六'], slots: TIME_SLOTS, data })
  }

  return fail(res, '接口不存在: ' + method + ' ' + path, 404)
```

```js
    await fetchList()

    return data.list
```

```vue
  return data.list.find(
    (r) =>
      (r.status === 'pending' || r.status === 'approved') &&
      form.startTime < r.endTime &&
      r.startTime < form.endTime

  } finally {

    // 拦截器已提示
```

### 5.3 步骤 3：新建 `src/stores/auth.js`

```js
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { login as loginApi } from '@/api/auth'

export const useAuthStore = defineStore('auth', () => {
  // 从 localStorage 恢复，刷新页面不掉登录
  const token = ref(localStorage.getItem('lab_token') || '')
  const user = ref(JSON.parse(localStorage.getItem('lab_user') || 'null'))

  const isLogin = computed(() => !!token.value)
  const name = computed(() => (user.value ? user.value.name : '未登录'))

  /** 登录成功后写入 store 和 localStorage */
  async function login(form) {
    const data = await loginApi(form)
    token.value = data.token
    user.value = data.user
    localStorage.setItem('lab_token', data.token)
    localStorage.setItem('lab_user', JSON.stringify(data.user))
    return data
  }

  /** 退出：清空所有痕迹 */
  function logout() {
    token.value = ''
    user.value = null
    localStorage.removeItem('lab_token')
    localStorage.removeItem('lab_user')
  }

  return { token, user, isLogin, name, login, logout }
})
```

## 6. 通用排错表

| 现象 | 原因 | 处理 |
|---|---|---|
| Failed to fetch dynamically imported module | 某个 .vue 编译失败 | 看终端 Vite 报错，它会指出文件；多半是空文件或标签没闭合 |
| getActivePinia() was called but there was no active Pinia | main.js 没注入 pinia，或放在 router 之后 | 按 1.7 的顺序改 |
| 接口全部 404 | vite.config.js 没注册 mockApiPlugin，或没重启 | 补插件 + 重启 dev server |
| 页面数据空白 | 请求发了但字段名对不上 | F12 → Network 看返回，对照 db.js 的字段名 |
| 图表空白但接口有数据 | 容器高度为 0 | 给 ChartBox 传 height，别在隐藏的 tab 里初始化 |
| 切 tab 后图表变形 | 隐藏时宽度是 0 | ChartBox 已用 ResizeObserver 处理，别自己删掉 |
| el-radio / el-radio-button 不生效 | Element Plus 2.6+ 用 value 替代 label | 按本文档的 :value 写法 |
| 分页点了没反应 | @current-change 没传函数 | 写成 @current-change="store.fetchList()" |
| PowerShell 里 curl 报错 | curl 是别名，& 要转义 | 用 curl.exe，URL 用单引号包起来 |
| Port 5173 is in use | 旧 dev server 没关 | 关掉旧终端，或用它提示的新端口 |
| 刷新深链 404（部署后） | history 模式需要服务器重写 | 演示统一用 npm run dev / npm run preview |

2. 仪表盘：念 4 个指标数字 → 趋势图切「近 30 天」（25 秒）
3. 实验室管理：新增一条 → 表格出现 → 编辑容量 → 删除（60 秒）
4. 预约管理：提交一个与已有预约重叠的时段 → 被拦，念出提示（30 秒）→ 换时段提交成功（20 秒）
5. 审核通过那条预约（15 秒）
6. 回首页：待审核数字减 1（20 秒）
7. 数据统计：4 张图快速划过，说一句热力图用途（25 秒）
8. 404 页兜一句"防呆处理"（5 秒）

### 7.3 README 模板

```markdown
# 实验室预约系统

## 技术栈
Vue 3 + Vite + Vue Router + Pinia + Element Plus + ECharts

## 运行
cd frontend
npm install
npm run dev
打开 http://localhost:5173

## 演示账号
admin / 123456（管理员）
teacher / 123456（教师）

## 功能
- 登录 + 路由守卫
- 实验室管理：搜索、分页、新增、编辑、删除
- 预约管理：提交、时段冲突校验、审核、状态流转
- 设备管理、用户管理（只读）
- 数据统计：趋势、使用率、状态占比、时段热力图

## 说明
后端接口由前端 mock（vite 中间件）提供，无需额外启动服务；重启 dev server 会重置数据。
```
